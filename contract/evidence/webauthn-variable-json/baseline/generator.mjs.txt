// Generate an auditable, unrolled bounded SHA-256 circuit. No generated account
// artifacts are touched. The witness supplies arithmetic hints, never authority.
import { writeFileSync } from 'node:fs';

const MAX = 256;
const BLOCKS = Math.ceil((MAX + 9) / 64);
const IV = [0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a,
  0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19];
const K = [
  0x428a2f98, 0x71374491, 0xb5c0fbcf, 0xe9b5dba5, 0x3956c25b, 0x59f111f1, 0x923f82a4, 0xab1c5ed5,
  0xd807aa98, 0x12835b01, 0x243185be, 0x550c7dc3, 0x72be5d74, 0x80deb1fe, 0x9bdc06a7, 0xc19bf174,
  0xe49b69c1, 0xefbe4786, 0x0fc19dc6, 0x240ca1cc, 0x2de92c6f, 0x4a7484aa, 0x5cb0a9dc, 0x76f988da,
  0x983e5152, 0xa831c66d, 0xb00327c8, 0xbf597fc7, 0xc6e00bf3, 0xd5a79147, 0x06ca6351, 0x14292967,
  0x27b70a85, 0x2e1b2138, 0x4d2c6dfc, 0x53380d13, 0x650a7354, 0x766a0abb, 0x81c2c92e, 0x92722c85,
  0xa2bfe8a1, 0xa81a664b, 0xc24b8b70, 0xc76c51a3, 0xd192e819, 0xd6990624, 0xf40e3585, 0x106aa070,
  0x19a4c116, 0x1e376c08, 0x2748774c, 0x34b0bcb5, 0x391c0cb3, 0x4ed8aa4a, 0x5b9cca4f, 0x682e6ff3,
  0x748f82ee, 0x78a5636f, 0x84c87814, 0x8cc70208, 0x90befffa, 0xa4506ceb, 0xbef9a3f7, 0xc67178f2,
];
const seq = (n, fn) => Array.from({ length: n }, (_, i) => fn(i));
const constWord = n => `Word { bits: [${seq(32, i => !!((n >>> i) & 1)).join(', ')}], value: ${n} }`;
const out = [`// GENERATED by experiments/webauthn-variable-json/generate.mjs. Experimental.
// Private length and witness advice are constrained, including SHA-256 padding.
pragma language_version >= 0.27.0;
import CompactStandardLibrary;
include "webauthn";
import WebAuthn<21>;

struct Word { bits: Vector<32, Boolean>; value: Uint<32>; }
export struct SplitWord { bits: Vector<32, Boolean>; carry: Uint<4>; }
export struct JsonAdvice {
  origin: Bytes<96>;
  origin_length: Uint<16>;
  prefix_length: Uint<16>;
}
witness split_word(value: Uint<36>): SplitWord;
witness inspect_json(data: Bytes<256>, length: Uint<16>): JsonAdvice;

export ledger expected_challenge: Bytes<32>;
export ledger enrolled_policy: Bytes<32>;
export ledger limited_verification: Boolean;
export ledger accepted: Bytes<32>;

constructor(challenge: Bytes<32>, policy: Bytes<32>, limited: Boolean) {
  expected_challenge = disclose(challenge);
  enrolled_policy = disclose(policy);
  limited_verification = disclose(limited);
}

pure circuit from_bits(bits: Vector<32, Boolean>): Word {
  const value = (${seq(32, i => `(bits[${i}] ? ${2 ** i} : 0)`).join(' + ')}) as Uint<32>;
  return Word { bits: bits, value: value };
}

circuit word(value: Uint<36>): Word {
  const hint = split_word(value);
  const result = from_bits(hint.bits);
  assert((value as Field) == (result.value as Field) + (hint.carry as Field) * 4294967296,
         "dishonest SHA word decomposition");
  return result;
}

pure circuit word_bytes(a: Word): Bytes<4> {
  return Bytes[${seq(4, j => `(${seq(8, i => `(a.bits[${(3 - j) * 8 + i}] ? ${2 ** i} : 0)`).join(' + ')}) as Uint<8>`).join(', ')}];
}

circuit bytes_word(a: Bytes<4>): Word {
  return word(((a[0] as Uint<36>) * 16777216 + (a[1] as Uint<36>) * 65536
            + (a[2] as Uint<36>) * 256 + (a[3] as Uint<36>)) as Uint<36>);
}
`];

for (const [name, rotations, shift] of [
  ['small0', [7, 18], 3], ['small1', [17, 19], 10],
  ['large0', [2, 13, 22], null], ['large1', [6, 11, 25], null],
]) {
  out.push(`pure circuit ${name}(a: Word): Word {\n  return from_bits([${seq(32, i => {
    const terms = rotations.map(n => `a.bits[${(i + n) % 32}]`);
    if (shift !== null) terms.push(i + shift < 32 ? `a.bits[${i + shift}]` : 'false');
    return `((${terms[0]} != ${terms[1]}) != ${terms[2]})`;
  }).join(', ')}]);\n}\n`);
}
out.push(`pure circuit choose(a: Word, b: Word, c: Word): Word {
  return from_bits([${seq(32, i => `(a.bits[${i}] ? b.bits[${i}] : c.bits[${i}])`).join(', ')}]);
}
pure circuit majority(a: Word, b: Word, c: Word): Word {
  return from_bits([${seq(32, i => `((a.bits[${i}] && b.bits[${i}]) || (c.bits[${i}] && (a.bits[${i}] || b.bits[${i}])))`).join(', ')}]);
}
`);
for (const n of [2, 4, 5]) {
  out.push(`circuit add${n}(${seq(n, i => `a${i}: Word`).join(', ')}): Word {
  return word((${seq(n, i => `(a${i}.value as Uint<36>)`).join(' + ')}) as Uint<36>);
}\n`);
}
out.push(`circuit sha_compress(initial: Vector<8, Word>, block: Bytes<64>): Vector<8, Word> {`);
for (let i = 0; i < 64; i++) {
  out.push(i < 16
    ? `  const w${i} = bytes_word(slice<4>(block, ${i * 4}));`
    : `  const w${i} = add4(small1(w${i - 2}), w${i - 7}, small0(w${i - 15}), w${i - 16});`);
}
out.push('  const s0 = initial;');
for (let i = 0; i < 64; i++) {
  out.push(`  const t${i} = add5(s${i}[7], large1(s${i}[4]), choose(s${i}[4], s${i}[5], s${i}[6]), ${constWord(K[i])}, w${i});
  const u${i} = add2(large0(s${i}[0]), majority(s${i}[0], s${i}[1], s${i}[2]));
  const s${i + 1} = [add2(t${i}, u${i}), s${i}[0], s${i}[1], s${i}[2], add2(s${i}[3], t${i}), s${i}[4], s${i}[5], s${i}[6]];`);
}
out.push(`  return [${seq(8, i => `add2(initial[${i}], s64[${i}])`).join(', ')}];\n}\n`);
out.push(`pure circuit padded_byte(input: Uint<8>, index: Uint<16>, length: Uint<16>, padded_length: Uint<16>, bit_length: Bytes<4>): Uint<8> {
  return index < length ? input
    : index == length ? 128
    : index == padded_length - 2 ? bit_length[2]
    : index == padded_length - 1 ? bit_length[3]
    : 0;
}

circuit sha256_bounded(data: Bytes<256>, length: Uint<16>): Bytes<32> {
  assert(length <= 256, "client data exceeds capacity");
  ${seq(MAX, i => `assert(length > ${i} || data[${i}] == 0, "nonzero buffer tail");`).join('\n  ')}
  const padded_length = (length <= 55 ? 64 : length <= 119 ? 128 : length <= 183 ? 192 : length <= 247 ? 256 : 320) as Uint<16>;
  const bit_length = word_bytes(word(((length as Uint<36>) * 8) as Uint<36>));
  const padded = Bytes[${seq(BLOCKS * 64, i => `padded_byte(${i < MAX ? `data[${i}]` : '0'}, ${i}, length, padded_length, bit_length)`).join(',\n    ')}];
  const h0 = [${IV.map(constWord).join(',\n    ')}];
  ${seq(BLOCKS, i => `const h${i + 1} = sha_compress(h${i}, slice<64>(padded, ${i * 64}));`).join('\n  ')}
  const h = length <= 55 ? h1 : length <= 119 ? h2 : length <= 183 ? h3 : length <= 247 ? h4 : h5;
  return Bytes[${seq(8, i => `...word_bytes(h[${i}])`).join(', ')}];
}

export pure circuit policy_commitment(rp_hash: Bytes<32>, origin: Bytes<96>, origin_length: Uint<16>, pk: Secp256r1Point): Bytes<32> {
  return persistentHash<[Bytes<32>, Bytes<96>, Uint<16>, Bytes<32>, Bytes<32>]>([
    rp_hash, origin, origin_length, secp256r1PointX(pk) as Bytes<32>, secp256r1PointY(pk) as Bytes<32>
  ]);
}

export circuit hash_bytes(data: Bytes<256>, length: Uint<16>): Bytes<32> {
  const digest = sha256_bounded(data, length);
  accepted = disclose(digest);
  return disclose(digest);
}

export circuit verify_webauthn(data: Bytes<256>, length: Uint<16>, rp_hash: Bytes<32>, authenticator_data: Bytes<37>, sig: Secp256r1EcdsaSignature, pk: Secp256r1Point): [] {
  assert(length <= 256, "client data exceeds capacity");
  const hint = inspect_json(data, length);
  assert(hint.origin_length > 0 && hint.origin_length <= 96, "origin length out of range");
  assert(hint.prefix_length == 112 + hint.origin_length, "dishonest prefix length");
  assert(length > hint.prefix_length, "truncated client data");
  assert(policy_commitment(rp_hash, hint.origin, hint.origin_length, pk) == enrolled_policy, "unknown key or origin policy");
  assert(slice<36>(data, 0) == pad(36, "{\\\"type\\\":\\\"webauthn.get\\\",\\\"challenge\\\":\\\""), "client data type/prefix mismatch");
  assert(slice<43>(data, 36) == webauthn_challenge_base64(expected_challenge), "challenge mismatch");
  assert(slice<12>(data, 79) == pad(12, "\\\",\\\"origin\\\":\\\""), "origin field missing");
  ${seq(96, i => `assert(${i} < hint.origin_length ? (hint.origin[${i}] == data[${91 + i}] && hint.origin[${i}] >= 33 && hint.origin[${i}] <= 126 && hint.origin[${i}] != 34 && hint.origin[${i}] != 92) : hint.origin[${i}] == 0, "origin encoding mismatch");`).join('\n  ')}
  const suffix = pad(21, "\\\",\\\"crossOrigin\\\":false");
  ${seq(96, i => `assert(hint.origin_length != ${i + 1} || slice<21>(data, ${92 + i}) == suffix, "cross-origin/suffix mismatch");`).join('\n  ')}
  const end_is_close = ${seq(96, i => `(hint.origin_length == ${i + 1} && data[${113 + i}] == 125)`).join(' || ')};
  const end_is_comma = ${seq(96, i => `(hint.origin_length == ${i + 1} && data[${113 + i}] == 44)`).join(' || ')};
  assert(end_is_close || end_is_comma, "client data boundary mismatch");
  if (!limited_verification) {
    assert(length == hint.prefix_length + 1 && end_is_close, "strict mode requires exact envelope");
  }
  assert(slice<32>(authenticator_data, 0) == rp_hash, "RP hash mismatch");
  const flags = authenticator_data[32];
  assert(flags == 5 || flags == 13 || flags == 29, "flags require UP and UV without extensions");
  const client_hash = sha256_bounded(data, length);
  const digest = persistentHash<[Bytes<37>, Bytes<32>]>([authenticator_data, client_hash]);
  assert(p256_verify_digest(digest, sig, pk), "invalid WebAuthn signature");
  accepted = disclose(client_hash);
}
`);
// Compact uses normal JSON-style string escapes, not literal backslashes.
const source = out.join('\n').replaceAll('\\\\\\"', '\\"');
writeFileSync(new URL('../../contract/contracts/probe-webauthn-variable.compact', import.meta.url), source);
console.log(`Generated bounded SHA-256 (${MAX} bytes, ${BLOCKS} blocks) and WebAuthn probes.`);
